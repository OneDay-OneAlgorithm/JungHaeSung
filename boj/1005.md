# ACM Craft
https://www.acmicpc.net/problem/1005

```cpp
#include <bits/stdc++.h>
using namespace std;
int N, K, W, A[1010], dp[1010];
vector<vector<int>> G;

int dfs(int v) {
	int& ret = dp[v];
	if (ret != -1) return ret;

	int m = 0;
	for (int u : G[v]) m = max(m, dfs(u));
	ret = A[v] + m;
	return ret;
}

void solve() {
	memset(dp, -1, sizeof dp);
	memset(A, 0, sizeof A);
	cin >> N >> K;
	G.clear();
	G.resize(N + 1);
	for (int i = 1; i <= N; i++) cin >> A[i];
	for (int i = 1, x, y; i <= K; i++) cin >> x >> y, G[y].push_back(x);
	cin >> W;
	cout << dfs(W) << '\n';
}

int main() {
	ios_base::sync_with_stdio(0);
	cin.tie(0);
	int T;
	cin >> T;
	while (T--) solve();
}
```
