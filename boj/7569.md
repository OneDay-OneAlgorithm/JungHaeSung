# 토마토
https://www.acmicpc.net/problem/7569

```cpp
#include <bits/stdc++.h>
using namespace std;
const tuple<int, int, int> dd[] = {
	{0, 0, 1}, {0, 0, -1}, {0, 1, 0}, {0, -1, 0}, {1, 0, 0}, {-1, 0, 0},
};
int M, N, H, A[101][101][101];
bool vis[101][101][101];

bool any_zero() {
	for (int i = 1; i <= H; i++)
		for (int j = 1; j <= N; j++)
			for (int k = 1; k <= M; k++)
				if (A[i][j][k] == 0) return true;
	return false;
}

int main() {
	ios_base::sync_with_stdio(0);
	cin.tie(0);
	cin >> M >> N >> H;
	queue<tuple<int, int, int, int>> q;
	for (int i = 1; i <= H; i++)
		for (int j = 1; j <= N; j++)
			for (int k = 1; k <= M; k++) {
				cin >> A[i][j][k];
				if (A[i][j][k] == 1) q.emplace(i, j, k, 0), vis[i][j][k] = true;
			}
	int ans = 0;
	while (!q.empty()) {
		auto [i, j, k, date] = q.front();
		q.pop();

		A[i][j][k] = 1;
		ans = max(ans, date);
		for (auto [di, dj, dk] : dd) {
			int ii = i + di, jj = j + dj, kk = k + dk;
			if (1 <= ii && ii <= H && 1 <= jj && jj <= N && 1 <= kk && kk <= M)
				if (!vis[ii][jj][kk] && A[ii][jj][kk] == 0) {
					vis[ii][jj][kk] = true;
					q.emplace(ii, jj, kk, date + 1);
				}
		}
	}
	cout << (any_zero() ? -1 : ans);
}
```
